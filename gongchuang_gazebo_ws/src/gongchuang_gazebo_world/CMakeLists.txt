cmake_minimum_required(VERSION 3.16)
project(gongchuang_gazebo_world)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(gongchuang_turntable SHARED src/turntable_system.cpp)
target_link_libraries(gongchuang_turntable PRIVATE gz-sim8::gz-sim8 gz-plugin2::gz-plugin2)
install(TARGETS gongchuang_turntable LIBRARY DESTINATION lib)
install(DIRECTORY worlds models config maps launch docs DESTINATION share/${PROJECT_NAME})
install(FILES README.md LICENSE generated_manifest.json requirements-build.txt DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/start.sh scripts/random_scene.sh scripts/capture.py scripts/generate.py scripts/runtime_check.py scripts/compare_map.py scripts/runtime_rules.py scripts/validate_bundle.py scripts/audit_slots.py
  DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/scene_details.py scripts/random_scenes.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_test(NAME geometry_maps_and_rules COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_geometry.py)
  add_test(NAME seeded_scenes_solvability COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_random_scenes.py)
endif()
ament_package()
